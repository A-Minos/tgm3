cmake_minimum_required(VERSION 3.22)
project(portlab_host C)
set(CMAKE_C_STANDARD 11)
add_library(guest_pages STATIC native/guest_pages.c native/guest_pages_selftest.c)
target_include_directories(guest_pages PUBLIC include)
target_compile_options(guest_pages PRIVATE -Wall -Wextra -Werror)
enable_testing()
foreach(case guest_pages randomized host_protection)
  add_executable(test_${case} tests/test_${case}.c)
  target_link_libraries(test_${case} guest_pages)
  add_test(NAME ${case} COMMAND test_${case})
endforeach()
add_executable(test_host_failures tests/test_host_failures.c)
target_link_libraries(test_host_failures guest_pages)
target_link_options(test_host_failures PRIVATE "-Wl,--wrap=mprotect" "-Wl,--wrap=madvise")
add_test(NAME host_failures COMMAND test_host_failures)
